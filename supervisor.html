// V5 Issue Tracker - server with permanent storage // Run: npm install express && node server.js const express = require('express'); const fs = require('fs'); const path = require('path'); const crypto = require('crypto'); const app = express(); app.use(express.json()); app.use(express.static(path.join(__dirname, 'public'))); // ---- Persistent storage (JSON file on disk) ---- // On a host like Render/Railway/Heroku, point DATA_DIR at a persistent disk/volume. const DATA_DIR = process.env.DATA_DIR || path.join(__dirname, 'data'); const DB_FILE = path.join(DATA_DIR, 'issues.json'); fs.mkdirSync(DATA_DIR, { recursive: true }); let issues = []; try { issues = JSON.parse(fs.readFileSync(DB_FILE, 'utf8')); } catch (e) { issues = []; // first run } function save() { // write to a temp file first, then rename, so a crash never corrupts data const tmp = DB_FILE + '.tmp'; fs.writeFileSync(tmp, JSON.stringify(issues, null, 2)); fs.renameSync(tmp, DB_FILE); } const dayKey = (iso) => new Date(iso).toLocaleDateString('en-CA', { timeZone: 'Asia/Kolkata' }); // YYYY-MM-DD // ---- API ---- // Advisor submits an issue app.post('/api/issues', (req, res) => { const { email, tlName, chatLink, issueFacing } = req.body || {}; if (!email || !tlName) return res.status(400).json({ error: 'email and tlName required' }); const issue = { id: crypto.randomUUID(), email: String(email).trim(), tlName: String(tlName).trim(), chatLink: String(chatLink || '').trim(), issueFacing: String(issueFacing || '').trim(), status: 'open', createdAt: new Date().toISOString(), fixedAt: null, }; issues.push(issue); save(); res.status(201).json(issue); }); // List issues (filters: status, date) app.get('/api/issues', (req, res) => { const { status, date } = req.query; let list = issues; if (status) list = list.filter((i) => i.status === status); if (date) list = list.filter((i) => dayKey(i.createdAt) === date); res.json([...list].sort((a, b) => b.createdAt.localeCompare(a.createdAt))); }); // Supervisor marks fixed app.patch('/api/issues/:id', (req, res) => { const issue = issues.find((i) => i.id === req.params.id); if (!issue) return res.status(404).json({ error: 'not found' }); if (req.body.status === 'fixed') { issue.status = 'fixed'; issue.fixedAt = new Date().toISOString(); } else if (req.body.status === 'open') { issue.status = 'open'; issue.fixedAt = null; } save(); res.json(issue); }); // Dashboard numbers app.get('/api/stats/summary', (req, res) => { const today = dayKey(new Date().toISOString()); const todays = issues.filter((i) => dayKey(i.createdAt) === today); res.json({ totalOpen: issues.filter((i) => i.status === 'open').length, openToday: todays.filter((i) => i.status === 'open').length, totalToday: todays.length, }); }); // CSV export (opens in Excel) app.get('/api/export', (req, res) => { const date = req.query.date; const list = date ? issues.filter((i) => dayKey(i.createdAt) === date) : issues; const q = (v) => `"${String(v ?? '').replace(/"/g, '""')}"`; const rows = [['Email', 'TL Name', 'Chat Link', 'Issue', 'Status', 'Created', 'Fixed']] .concat(list.map((i) => [i.email, i.tlName, i.chatLink, i.issueFacing, i.status, i.createdAt, i.fixedAt || ''])); res.setHeader('Content-Type', 'text/csv; charset=utf-8'); res.setHeader('Content-Disposition', `attachment; filename="v5-issues-${date || 'all'}.csv"`); res.send('\uFEFF' + rows.map((r) => r.map(q).join(',')).join('\n')); }); // Pages app.get('/supervisor', (req, res) => res.sendFile(path.join(__dirname, 'public', 'supervisor.html'))); const PORT = process.env.PORT || 3000; app.listen(PORT, () => console.log('Running on port ' + PORT));
